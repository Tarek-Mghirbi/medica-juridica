<script>
  // ============ TRILINGUAL SWITCHER ============
  // Simply add data-lang attributes (or data-fr / data-ar / data-en) to any element
  // to make it translateable. Elements without a translation stay unchanged.
  var available = ['fr','ar','en'];
  var current = 'fr';

  function setLang(lang){
    if(available.indexOf(lang)===-1) return;
    current = lang;
    // update buttons
    available.forEach(function(l){
      document.getElementById('btn-'+l).classList.toggle('active', l===lang);
    });
    // update html dir for arabic (RTL)
    document.documentElement.setAttribute('lang', lang);
    document.body.setAttribute('dir', lang==='ar' ? 'rtl' : 'ltr');
    // update text on all elements carrying translations
    document.querySelectorAll('[data-en]').forEach(function(el){
      var key = 'data-'+lang;
      if(el.hasAttribute(key)){
        el.textContent = el.getAttribute(key);
      }
    });
    // persist choice
    try{ localStorage.setItem('mj-lang', lang); }catch(e){}
  }

  // restore saved language on load
  (function(){
    var saved = null;
    try{ saved = localStorage.getItem('mj-lang'); }catch(e){}
    if(saved && available.indexOf(saved)!==-1){ setLang(saved); }
  })();
</script>
